        // ── Helper: add labeled field ─────────────────────────────
        private void AddField(Panel parent, string labelText, int y, ref TextBox tb, int width)
        {
            parent.Controls.Add(MakeLabel(labelText, new Point(20, y)));
            tb = MakeTextBox(new Point(20, y + 22), width);
            parent.Controls.Add(tb);
        }

        private Label MakeLabel(string text, Point loc)
        {
            return new Label
            {
                Text = text,
                Font = new Font("Segoe UI", 8f),
                ForeColor = Color.FromArgb(100, 100, 100),
                Location = loc,
                AutoSize = true
            };
        }

        private TextBox MakeTextBox(Point loc, int width)
        {
            TextBox tb = new TextBox
            {
                Location = loc,
                Width = width,
                Height = 30,
                Font = new Font("Segoe UI", 10.5f),
                BorderStyle = BorderStyle.FixedSingle,
                BackColor = Color.FromArgb(252, 252, 250)
            };
            tb.TextChanged += (s, e) => UpdatePreview();
            return tb;
        }

        private void UpdatePreview()

        // ── Preview Paint: Khaadi-style label ────────────────────
        private void PreviewPanel_Paint(object sender, PaintEventArgs e)
        {
            if (txtBrand == null || txtProduct == null || txtPrice == null || txtBarcode == null)
                return;

            Graphics g = e.Graphics;
            g.SmoothingMode = System.Drawing.Drawing2D.SmoothingMode.AntiAlias;

            int W = previewPanel.Width;
            int H = previewPanel.Height;

            // White background
            g.FillRectangle(Brushes.White, 0, 0, W, H);

            // Outer border
            using (Pen bp = new Pen(Color.FromArgb(180, 180, 180), 1.5f))
                g.DrawRectangle(bp, new Rectangle(1, 1, W - 2, H - 2));

            string brand = string.IsNullOrWhiteSpace(txtBrand.Text) ? "BRAND NAME" : txtBrand.Text.ToUpper();
            string product = string.IsNullOrWhiteSpace(txtProduct.Text) ? "PRODUCT" : txtProduct.Text.ToUpper();
            string price = string.IsNullOrWhiteSpace(txtPrice.Text) ? "0.00" : txtPrice.Text;
            string sku = string.IsNullOrWhiteSpace(txtBarcode.Text) ? "SKU-000000" : txtBarcode.Text;

            Font brandFont = new Font("Arial", 14f, FontStyle.Bold);
            Font skuFont = new Font("Arial", 7.5f);
            Font priceFont = new Font("Arial", 8f, FontStyle.Bold);
            Font productFont = new Font("Arial", 7.5f, FontStyle.Bold);

            int y = 30;

            // ── Brand Name ──────────────────────────
            SizeF bs = g.MeasureString(brand, brandFont);
            g.DrawString(brand, brandFont, Brushes.Black, (W - bs.Width) / 2, y);
            y += (int)bs.Height + 2;

            // ── SKU ─────────────────────────────────
            SizeF ss = g.MeasureString(sku, skuFont);
            g.DrawString(sku, skuFont, Brushes.Black, (W - ss.Width) / 2, y);
            y += (int)ss.Height + 4;

            // ── Barcode ─────────────────────────────
            int barcodeHeight = 55; // ✅ Controlled height
            if (barcodeImage != null)
            {
                g.DrawImage(barcodeImage, new Rectangle(8, y, W - 16, barcodeHeight));
            }
            else
            {
                g.FillRectangle(new SolidBrush(Color.FromArgb(245, 245, 245)),
                    new Rectangle(8, y, W - 16, barcodeHeight));
                using (Pen ph = new Pen(Color.FromArgb(200, 200, 200), 1))
                    g.DrawRectangle(ph, new Rectangle(8, y, W - 16, barcodeHeight));
                SizeF ps2 = g.MeasureString("[ barcode here ]", skuFont);
                g.DrawString("[ barcode here ]", skuFont,
                    new SolidBrush(Color.FromArgb(180, 180, 180)),
                    (W - ps2.Width) / 2, y + 18);
            }
            y += barcodeHeight + 6;

            // ── Separator Line ───────────────────────
            using (Pen sep = new Pen(Color.FromArgb(200, 200, 200), 1))
                g.DrawLine(sep, 8, y, W - 8, y);
            y += 5;

            // ── Price (left) + Product (right) ───────
            string priceStr = "PKR  " + price;
            g.DrawString(priceStr, priceFont, Brushes.Black, 10, y);
            SizeF prds = g.MeasureString(product, productFont);
            g.DrawString(product, productFont, Brushes.Black, W - prds.Width - 10, y + 1);
        }

        // ── Generate ─────────────────────────────────────────────
        private void BtnGenerate_Click(object sender, EventArgs e)
        {
            if (string.IsNullOrWhiteSpace(txtBarcode.Text))
            {
                ShowMsg("Please enter a Barcode / SKU value.", false);
                return;
            }

            var writer = new BarcodeWriter
            {
                Format = BarcodeFormat.CODE_128,
Options = new ZXing.Common.EncodingOptions
{
    Width = 500,
    Height = 70,   // ✅ 130 tha, ab 70 kar diya
    Margin = 0,
    PureBarcode = true
}
            };

            barcodeImage = writer.Write(txtBarcode.Text);
            //picBarcode.Image = barcodeImage;
            previewPanel.Invalidate();
            ShowMsg("Barcode generated!", true);
        }

        // ── Print ────────────────────────────────────────────────
        private void BtnPrint_Click(object sender, EventArgs e)
        {
            if (barcodeImage == null) { ShowMsg("Generate barcode first.", false); return; }
            if (!int.TryParse(txtQty.Text.Trim(), out int qty) || qty < 1)
            { ShowMsg("Enter valid quantity (min 1).", false); return; }

            SetupPrintDoc(qty);
            using (PrintDialog pd = new PrintDialog())
            {
                pd.Document = printDoc;
                pd.UseEXDialog = true;
                if (pd.ShowDialog() == DialogResult.OK)
                {
                    printDoc.Print();
                    ShowMsg(qty + " label" + (qty == 1 ? "" : "s") + " sent to printer!", true);
                }
            }
        }

        // ── Preview ──────────────────────────────────────────────
        private void BtnPreview_Click(object sender, EventArgs e)
        {
            if (barcodeImage == null) { ShowMsg("Generate barcode first.", false); return; }
            if (!int.TryParse(txtQty.Text.Trim(), out int qty) || qty < 1) qty = 1;

            SetupPrintDoc(qty);
            PrintPreviewDialog prev = new PrintPreviewDialog
            {
                Document = printDoc,
                Width = 700,
                Height = 520,
                Text = "Print Preview  -  " + qty + " cop" + (qty == 1 ? "y" : "ies")
            };
            prev.ShowDialog();
        }

        // ── Shared print setup ───────────────────────────────────
        private void SetupPrintDoc(int qty)
        {
            printDoc = new PrintDocument();
            PaperSize ps = new PaperSize("LabelSize", 150, 120);
            printDoc.DefaultPageSettings.PaperSize = ps;
            printDoc.DefaultPageSettings.Margins = new Margins(2, 0, 0, 0);

            int copies = qty, currentCopy = 0;
            printDoc.PrintPage += (s, ev) =>
            {
                PrintLabelPage(ev);
                currentCopy++;
                ev.HasMorePages = currentCopy < copies;
            };
        }

        // ── Actual print layout (1.5x1.2 inch) ──────────────────
        private void PrintLabelPage(PrintPageEventArgs e)
        {
            Graphics g = e.Graphics;
            g.SmoothingMode = System.Drawing.Drawing2D.SmoothingMode.AntiAlias;

            Font brandFont = new Font("Arial", 11f, FontStyle.Bold);
            Font skuFont = new Font("Arial", 6.5f);
            Font priceFont = new Font("Arial", 6.5f, FontStyle.Bold);
            Font productFont = new Font("Arial", 5.5f);

            string brand = txtBrand.Text.ToUpper();
            string sku = txtBarcode.Text;
            string price = "PKR " + txtPrice.Text;
            string product = txtProduct.Text.ToUpper();

            int W = 150; // ✅ Apni label width
            int y = 5;   // ✅ Top margin

            // ── Brand Name ──────────────────────────────
            SizeF brandSize = g.MeasureString(brand, brandFont);
            g.DrawString(brand, brandFont, Brushes.Black, (W - brandSize.Width) / 2, y);
            y += (int)brandSize.Height + 1;

            // ── SKU ─────────────────────────────────────
            SizeF skuSize = g.MeasureString(sku, skuFont);
            g.DrawString(sku, skuFont, Brushes.Black, (W - skuSize.Width) / 2, y);
            y += (int)skuSize.Height + 2;

            // ── Barcode ─────────────────────────────────
            if (barcodeImage != null)
            {
                g.DrawImage(barcodeImage, new Rectangle(5, y, W - 4, 40)); // ✅ Height 38
                y += 41;
            }

            // ── Separator Line ───────────────────────────
            using (Pen sep = new Pen(Color.FromArgb(150, 150, 150), 0.5f))
                g.DrawLine(sep, 5, y, W - 5, y); // ✅ Line full width
            y += 4;

            // ── Price (Left) + Product (Right) ───────────
            g.DrawString(price, priceFont, Brushes.Black, 5, y);
            SizeF prodSize = g.MeasureString(product, productFont);
            g.DrawString(product, productFont, Brushes.Black, W - prodSize.Width - 5, y + 1);
        }

        private void ShowMsg(string msg, bool success)
        {
            MessageBox.Show(msg, success ? "Done" : "Notice",
                MessageBoxButtons.OK,
                success ? MessageBoxIcon.Information : MessageBoxIcon.Warning);
        }
